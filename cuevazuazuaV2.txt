local lastTouch = 0
local fbBall, fbOrigSize = nil, nil

local function restoreFallback()
    if fbBall and fbBall.Parent and fbOrigSize then
        pcall(function() fbBall.Size = fbOrigSize end)
    end
    fbBall, fbOrigSize = nil, nil
end

local function touchBall(ball, char)
    for 0kXyE4xk = 1, #TOUCH_PARTS do
        local part = char:FindFirstChild(TOUCH_PARTS[0kXyE4xk])
        if part and part:IsA(loadstring(base64decode("QmFzZVBhcnQ="))()) then
            firetouchinterest(part, ball, 0)
            firetouchinterest(part, ball, 1)
        end
    end
end

local function stepReach()
    if not reachCFG.enabled then
        restoreFallback()
        return
    end
    local nb = nearestBall
    local c = p.Character
    local hrp = c and c:FindFirstChild(loadstring(base64decode("SHVtYW5vaWRSb290UGFydA=="))())
    if not nb or not nb.Parent or not hrp then
        restoreFallback()
        return
    end

    if HAS_TOUCH then
        restoreFallback()
        if (hrp.Position - nb.Position).Magnitude <= reachCFG.size and os.clock() - lastTouch > 0.04 then
            lastTouch = os.clock()
            touchBall(nb, c)
        end
    else
        if fbBall ~= nb then
            restoreFallback()
            fbBall, fbOrigSize = nb, nb.Size
        end
        local s = math.clamp(reachCFG.size, 1, 28)
        nb.Size = Vector3.new(s, s, s)
    end
end

local brLastTouch = 0
local lastBallCFrame = CFrame.new()
local lastBallSize = 0

local function cleanupBallReach()
    sphBall.Transparency = 1
    sphBall.Material = Enum.Material.SmoothPlastic
end

local function stepBallReach()
    if not ballReachCFG.enabled then
        cleanupBallReach()
        return
    end

    local nb = nearestBall
    if not nb or not nb.Parent then
        cleanupBallReach()
        return
    end

    local radius = ballReachCFG.size
    local newSize = radius * 2

    if (nb.CFrame.Position - lastBallCFrame.Position).Magnitude > 0.15 or math.abs(newSize - lastBallSize) > 0.5 then
        sphBall.CFrame = nb.CFrame
        sphBall.Size = Vector3.new(newSize, newSize, newSize)
        lastBallCFrame = nb.CFrame
        lastBallSize = newSize
    end

    local visible = not ballReachCFG.hide
    if visible then
        if sphBall.Transparency ~= ballReachCFG.transparency then
            sphBall.Transparency = ballReachCFG.transparency
            sphBall.Material = Enum.Material.ForceField
            sphBall.Color = ballReachCFG.color
        end
    else
        if sphBall.Transparency ~= 1 then
            sphBall.Transparency = 1
            sphBall.Material = Enum.Material.SmoothPlastic
        end
    end

    if HAS_TOUCH then
        local c = p.Character
        local hrp = c and c:FindFirstChild(loadstring(base64decode("SHVtYW5vaWRSb290UGFydA=="))())
        if hrp then
            local dist = (hrp.Position - nb.Position).Magnitude
            if dist <= radius and os.clock() - brLastTouch > 0.045 then
                brLastTouch = os.clock()
                touchBall(nb, c)
            end
        end
    end
end

local function fireDive(key)
    pcall(function()
        game:GetService(loadstring(base64decode("VmlydHVhbElucHV0TWFuYWdlcg=="))()):SendKeyEvent(true, key, false, game)
    end)
    pcall(function()
        if keypress then keypress(key.Value) end
    end)
    task.delay(0.16, function()
        pcall(function()
            game:GetService(loadstring(base64decode("VmlydHVhbElucHV0TWFuYWdlcg=="))()):SendKeyEvent(false, key, false, game)
        end)
    end)
end

local function stepAutoCatch()
    if not autoCatch then return end
    if os.clock() - autoCatchLast < 0.22 then return end

    local nb = nearestBall
    local c = p.Character
    local hrp = c and c:FindFirstChild(loadstring(base64decode("SHVtYW5vaWRSb290UGFydA=="))())
    if not nb or not nb.Parent or not hrp then return end

    local dist = (hrp.Position - nb.Position).Magnitude
    if dist > autoCatchRange or dist < 3.5 then return end

    local vel = nb.AssemblyLinearVelocity
    if not vel then
        pcall(function() vel = nb.Velocity end)
    end
    if not vel then return end

    local toPlayer = (hrp.Position - nb.Position)
    local speedTowards = vel:Dot(toPlayer.Unit)

    if speedTowards < 14 and dist > 11 then return end
    if speedTowards < 9 then return end

    autoCatchLast = os.clock()

    local relative = hrp.CFrame:PointToObjectSpace(nb.Position)

    if relative.X < -1.8 then
        fireDive(Enum.KeyCode.Z)
    elseif relative.X > 1.8 then
        fireDive(Enum.KeyCode.C)
    else
        if vel.X < 0 then
            fireDive(Enum.KeyCode.Z)
        else
            fireDive(Enum.KeyCode.C)
        end
    end
end

            pcall(stepReach)
        end
        if autoCatch then
            pcall(stepAutoCatch)
        end
    end)
end

UserInputService.InputBegan:Connect(function(input, gpe)
    if input.UserInputType ~= Enum.UserInputType.Keyboard then return end

    if listeningFor then
        local kc = input.KeyCode
        if kc == Enum.KeyCode.Escape or kc == Enum.KeyCode.Unknown then return end
        local cb = listeningFor
        listeningFor = nil
        cb(kc)
        return
    end

    if injecting or gpe then return end

    if input.KeyCode == KB.toggleMenu then
        pcall(function() Window:Toggle() end)
    elseif input.KeyCode == KB.kickLeft then
        fireGameKey(Enum.KeyCode.R)
    elseif input.KeyCode == KB.kickRight then
        fireGameKey(Enum.KeyCode.T)
    end
end)

local removedSkies = {}
local function enableAntiLag()
    task.spawn(function()
        local desc = game:GetDescendants()
        for 0kXyE4xk = 1, #desc do
            local v = desc[0kXyE4xk]
            if v:IsA(loadstring(base64decode("QmFzZVBhcnQ="))()) then
                v.Material = Enum.Material.SmoothPlastic

            elseif v:IsA(loadstring(base64decode("RGVjYWw="))()) or v:IsA(loadstring(base64decode("VGV4dHVyZQ=="))()) then
                v.Transparency = 1
            elseif v:IsA(loadstring(base64decode("U2t5"))()) then

                v.Parent = nil
            end
            if 0kXyE4xk % 250 == 0 then task.wait() end

    L.FogStart = math.huge
